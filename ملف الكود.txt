        // Category Filtering
        function filterCategory(categoryName) {
            const buttons = document.querySelectorAll('.cat-btn');
            buttons.forEach(btn => {
                if (btn.getAttribute('data-category') === categoryName) {
                    btn.className = "cat-btn active px-5 py-2.5 rounded-xl text-xs sm:text-sm font-bold bg-arafa-600 text-white shadow-md transition";
                } else {
                    btn.className = "cat-btn px-5 py-2.5 rounded-xl text-xs sm:text-sm font-bold bg-white text-slate-700 hover:bg-slate-100 border border-slate-200 transition";
                }
            });

            if (categoryName === 'all') {
                displayProducts(arafaProducts);
            } else {
                const filtered = arafaProducts.filter(p => p.category === categoryName);
                displayProducts(filtered);
            }
        }

        // Installment Calculator logic
        const priceSlider = document.getElementById('priceRange');
        const downSlider = document.getElementById('downRange');
        let selectedInstallmentMonths = 12;

        function refreshCalculator() {
            let price = parseInt(priceSlider.value);
            let down = parseInt(downSlider.value);

            if (down >= price) {
                down = price - 2000;
                downSlider.value = down;
            }

            document.getElementById('calcPriceLabel').innerText = price.toLocaleString('ar-EG') + ' ج.م';
            document.getElementById('calcDownLabel').innerText = down.toLocaleString('ar-EG') + ' ج.م';

            const remaining = price - down;
            document.getElementById('calcRemainingResult').innerText = remaining.toLocaleString('ar-EG') + ' ج.م';

            // Approx simple flat interest calculation (~12% per year)
            const annualInterest = 0.12;
            const totalInterest = remaining * (annualInterest * (selectedInstallmentMonths / 12));
            const totalWithInterest = remaining + totalInterest;
            const monthlyPayment = Math.round(totalWithInterest / selectedInstallmentMonths);

            document.getElementById('calcMonthlyResult').innerText = monthlyPayment.toLocaleString('ar-EG');
        }

        function setCalcMonths(m) {
            selectedInstallmentMonths = m;
            document.querySelectorAll('.month-pill').forEach(btn => {
                if (parseInt(btn.getAttribute('data-val')) === m) {
                    btn.className = "month-pill active py-2.5 text-xs font-bold rounded-xl border bg-arafa-600 text-white border-arafa-600";
                } else {
                    btn.className = "month-pill py-2.5 text-xs font-bold rounded-xl border border-slate-300 text-slate-700 hover:border-arafa-600";
                }
            });
            refreshCalculator();
        }
